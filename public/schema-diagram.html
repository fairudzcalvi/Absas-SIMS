📊 ABSAS SIMS Database Schema

Multi-Year Calendar, SHS Strands, Dynamic Sections, Faculty, Student Lifecycle, and Finance Ledger

erDiagram
    SCHOOL_YEARS {
        uuid id PK
        varchar year_label "Unique e.g. 2025-2026"
        date start_date
        date end_date
        boolean is_active "Active flag"
    }

    QUARTERS {
        uuid id PK
        uuid school_year_id FK "References school_years"
        int quarter_number "1 to 4"
        varchar quarter_name "1st to 4th Quarter"
        boolean is_active "Active quarter flag"
        boolean is_grading_open "Lock or open grading"
    }

    SHS_STRANDS {
        uuid id PK
        varchar strand_code "STEM, ABM, HUMSS, TVL"
        varchar strand_name "Strand Name"
        varchar track_type "Academic or TVL"
        varchar status "Active or Inactive"
    }

    SECTIONS {
        uuid id PK
        int grade_level "Grades 1 to 12"
        varchar section_name "e.g. Diamond, Emerald"
        uuid school_year_id FK "References school_years"
        uuid strand_id FK "References shs_strands"
        bigint adviser_id FK "References faculty"
        varchar room_number
        int max_capacity "Max 45"
    }

    SUBJECTS {
        uuid subject_id PK
        varchar subject_code "e.g. MATH-JH, SCI-11"
        varchar subject_name "Subject Title"
        varchar department "Elementary, JHS, SHS"
        int grade_level "Grade 1 to 12"
        varchar semester "1st or 2nd Sem"
        uuid strand_id FK "References shs_strands"
    }

    SCHEDULES {
        uuid id PK
        uuid section_id FK "References sections"
        uuid subject_id FK "References subjects"
        bigint faculty_id FK "References faculty"
        uuid school_year_id FK "References school_years"
        uuid quarter_id FK "References quarters"
        varchar schedule_day "Day of week"
        time start_time
        time end_time
    }

    ADMINS {
        uuid id PK
        varchar username
        varchar full_name
        varchar email
        varchar role "SuperAdmin, Registrar, Cashier"
    }

    FACULTY {
        bigint faculty_record_id PK
        varchar employee_id "Employee Code"
        varchar first_name
        varchar last_name
        varchar email
        varchar department "Elementary, JHS, SHS"
        varchar position "Principal, Head, Teacher"
        int adviser_grade_level
        boolean is_archived
    }

    STUDENTS {
        bigint student_record_id PK
        varchar lrn "12-digit DepEd LRN"
        varchar first_name
        varchar last_name
        varchar gender
        date birth_date
        int grade_level "Grades 1 to 12"
        uuid current_section_id FK "References sections"
        uuid current_strand_id FK "References shs_strands"
        uuid scholarship_id FK "References scholarships"
        varchar status "Active, Graduated, Transferred"
        boolean is_archived
    }

    ENROLLMENTS {
        uuid id PK
        bigint student_record_id FK "References students"
        uuid school_year_id FK "References school_years"
        uuid section_id FK "References sections"
        uuid strand_id FK "References shs_strands"
        uuid scholarship_id FK "References scholarships"
        varchar enrollment_type "New, Continuing, Transferee"
        varchar enrollment_status "Enrolled, Pending, Dropped"
        jsonb requirements_submitted "PSA, Form 138, etc"
        date date_enrolled
    }

    GRADES {
        bigint id PK
        bigint student_record_id FK "References students"
        varchar subject "Subject code"
        varchar quarter "1st, 2nd, 3rd, 4th"
        numeric written_works "WW score"
        numeric performance_tasks "PT score"
        numeric quarterly_assessment "QA score"
        numeric quarter_grade "Transmuted Grade"
        varchar remarks "Passed or Failed"
    }

    ATTENDANCE {
        bigint id PK
        bigint student_record_id FK "References students"
        date date
        varchar status "Present, Absent, Late"
        varchar remarks
    }

    TRANSCRIPTS {
        bigint id PK
        bigint student_record_id FK "References students"
        uuid school_year_id FK "References school_years"
        numeric general_average "Final GPA"
        varchar promotional_status "Promoted or Retained"
        jsonb quarterly_grades_snapshot
    }

    SCHOLARSHIPS {
        uuid id PK
        varchar name "ESC Voucher, Honor, Sibling"
        varchar code "ESC-GRANT, SIBLING-10"
        varchar discount_type "percentage or fixed"
        numeric discount_value "Amount or percent"
        boolean is_active
    }

    STUDENT_FINANCES {
        bigint id PK
        bigint student_record_id FK "References students"
        uuid school_year_id FK "References school_years"
        uuid scholarship_id FK "References scholarships"
        numeric tuition_fee "Tuition assessment"
        numeric discount_amount "Total discount"
        numeric net_payable "Net due"
        numeric total_paid "Total payments"
        numeric balance "Remaining balance"
        varchar payment_status "Paid, Partial, Unpaid"
    }

    PAYMENTS {
        bigint payment_id PK
        bigint student_record_id FK "References students"
        uuid school_year_id FK "References school_years"
        varchar or_number "Official Receipt Number"
        numeric amount "Payment amount"
        varchar payment_method "Cash, GCash, Bank"
        varchar payment_for "Tuition or Other Fees"
        varchar cashier_name
        timestamp payment_date
        varchar reference_number
    }

    SCHOOL_YEARS ||--o{ QUARTERS : "manages"

    SCHOOL_YEARS ||--o{ ENROLLMENTS : "tracks"
    SCHOOL_YEARS ||--o{ STUDENT_FINANCES : "bills"

    SHS_STRANDS ||--o{ SECTIONS : "categorizes"
    FACULTY ||--o{ SECTIONS : "advises"

    SECTIONS ||--o{ ENROLLMENTS : "assigns"
    SECTIONS ||--o{ STUDENTS : "places"
    SECTIONS ||--o{ SCHEDULES : "timetables"

    SUBJECTS ||--o{ SCHEDULES : "taught_in"
    FACULTY ||--o{ SCHEDULES : "instructs"

    STUDENTS ||--o{ ENROLLMENTS : "enrolls_in"
    STUDENTS ||--o{ GRADES : "earns"
    STUDENTS ||--o{ ATTENDANCE : "logged_in"
    STUDENTS ||--o{ TRANSCRIPTS : "records"
    STUDENTS ||--|| STUDENT_FINANCES : "ledger_for"
    STUDENTS ||--o{ PAYMENTS : "receipted_for"

    SCHOLARSHIPS ||--o{ STUDENTS : "granted_to"
    SCHOLARSHIPS ||--o{ STUDENT_FINANCES : "deducted_from"
    STUDENT_FINANCES ||--o{ PAYMENTS : "credited_by"
